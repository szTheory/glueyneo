set(GLUEYNEO_CPU_SANITIZER NONE CACHE STRING "NONE, ADDRESS_UNDEFINED or THREAD")
set(GLUEYNEO_CPU_OPTIMIZATION NONE CACHE STRING "NONE or RELEASE")
if(NOT GLUEYNEO_CPU_SANITIZER MATCHES "^(NONE|ADDRESS_UNDEFINED|THREAD)$")
  message(FATAL_ERROR "Invalid CPU sanitizer")
endif()
if(NOT GLUEYNEO_CPU_OPTIMIZATION MATCHES "^(NONE|RELEASE)$")
  message(FATAL_ERROR "Invalid CPU optimization")
endif()
add_executable(cpu_generator ../../third_party/musashi/m68kmake.c)
target_compile_features(cpu_generator PRIVATE c_std_17)
add_library(cpu_runtime STATIC cpu_adapter.c ../../third_party/musashi/m68kcpu.c ../../third_party/musashi/m68kops.c)
target_include_directories(cpu_runtime PRIVATE ../../third_party/musashi)
target_include_directories(cpu_runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(cpu_runtime PRIVATE c_std_17)
add_library(cpu_unity STATIC ../../third_party/unity/src/unity.c)
target_include_directories(cpu_unity PUBLIC ../../third_party/unity/src)
add_executable(cpu_guest test_bus.c ../../tests/cpu/test_guest.c ../../tests/cpu/guest_fixture.c)
target_link_libraries(cpu_guest PRIVATE cpu_runtime cpu_unity)
foreach(target cpu_runtime cpu_guest cpu_unity)
  target_compile_features(${target} PRIVATE c_std_17)
  if(GLUEYNEO_CPU_OPTIMIZATION STREQUAL RELEASE)
    target_compile_options(${target} PRIVATE -O2)
  else()
    target_compile_options(${target} PRIVATE -O0)
  endif()
  if(GLUEYNEO_CPU_SANITIZER STREQUAL ADDRESS_UNDEFINED)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  elseif(GLUEYNEO_CPU_SANITIZER STREQUAL THREAD)
    target_compile_options(${target} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=thread)
  endif()
endforeach()
add_test(NAME cpu_guest COMMAND cpu_guest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME cpu_guest_negative COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/cpu/negative.py $<TARGET_FILE:cpu_guest>)
set_tests_properties(cpu_guest cpu_guest_negative PROPERTIES LABELS "cpu;cpu-fast" TIMEOUT 30)
